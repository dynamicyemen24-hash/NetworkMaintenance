التكامل مع Stripe

import stripe
stripe.api_key = os.getenv("STRIPE_SECRET_KEY")
# POST /api/v1/subscription/checkout → stripe.checkout.Session.create(...)
# POST /api/v1/subscription/webhook → stripe.Webhook.construct_event(...)
